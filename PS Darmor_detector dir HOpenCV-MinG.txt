cv::Mat ArmorDetector::preprocess(const cv::Mat& src) {
    cv::Mat blurred, hsv_image;

    // 鏍规嵁鍥惧儚澶у皬璋冩暣楂樻柉妯＄硦鏍稿ぇ灏?
    int kernel_size = std::max(3, std::min(src.cols, src.rows) / 200);
    if (kernel_size % 2 == 0) kernel_size++; // 纭繚鏍稿ぇ灏忎负濂囨暟

    // 楂樻柉妯＄硦鍘诲櫔
    cv::GaussianBlur(src, blurred, cv::Size(kernel_size, kernel_size), 0);

    // 杞崲鍒癏SV棰滆壊绌洪棿
    cv::cvtColor(blurred, hsv_image, cv::COLOR_BGR2HSV);

    return hsv_image;
}

// 鐏潯鐗瑰緛绛涢€?- 鍙傝€僐oboMaster 2018 SEU妗堜緥杩涜浼樺寲
bool ArmorDetector::isLightBar(const std::vector<cv::Point>& contour) {
    double area = cv::contourArea(contour);
    if (area < 50 || area > 5000) return false; // 闈㈢Н绛涢€?
    cv::RotatedRect rect = cv::minAreaRect(contour);
    float width = rect.size.width;
    float height = rect.size.height;
    if (width < height) std::swap(width, height);
    float aspect_ratio = width / height;
    // 璋冩暣闀垮姣旇寖鍥达紝鍙傝€僐oboMaster 2018妗堜緥
    if (aspect_ratio < 2.5 || aspect_ratio > 12) return false; // 闀垮姣旂瓫閫?
    return true;
}

// 瑁呯敳鏉块厤瀵?- 鍙傝€僐oboMaster 2018 SEU妗堜緥杩涜浼樺寲
std::vector<cv::Point2f> ArmorDetector::pairLightBars(const std::vector<cv::RotatedRect>& light_bars) {
    std::vector<cv::Point2f> armor_centers;
    for (size_t i = 0; i < light_bars.size(); i++) {
        for (size_t j = i + 1; j < light_bars.size(); j++) {
            cv::Point2f center1 = light_bars[i].center;
            cv::Point2f center2 = light_bars[j].center;
            cv::RotatedRect rect1 = light_bars[i];
            cv::RotatedRect rect2 = light_bars[j];

            // 璁＄畻鐏潯楂樺害
            float height1 = std::max(rect1.size.width, rect1.size.height);
            float height2 = std::max(rect2.size.width, rect2.size.height);
            float min_height = std::min(height1, height2);

            // 璁＄畻涓や釜鐏潯涔嬮棿鐨勮窛绂?
            float distance = cv::norm(center1 - center2);

            // 璺濈搴斾笌鐏潯楂樺害鎴愭瘮渚?
            if (distance < 2 * min_height || distance > 8 * min_height) {
                continue;
            }

            // 璁＄畻瑙掑害宸紙鑰冭檻OpenCV涓搴︾殑鐗规畩鎬э級
            float angle1 = rect1.angle;
            float angle2 = rect2.angle;
            if (angle1 < -45) angle1 += 90;
            if (angle2 < -45) angle2 += 90;
            float angle_diff = std::abs(angle1 - angle2);
            if (angle_diff > 15) {
                continue;
            }

            // 璁＄畻姘村钩鍜屽瀭鐩磋窛绂?
            float horizontal_dist = std::abs(center1.x - center2.x);
            float vertical_dist = std::abs(center1.y - center2.y);
            // 姘村钩璺濈搴斿ぇ浜庡瀭鐩磋窛绂伙紙瑁呯敳鏉块€氬父鏄按骞虫帓鍒楋級
            if (horizontal_dist < vertical_dist) {
                continue;
            }

            // 璁＄畻瑁呯敳鏉夸腑蹇?
            cv::Point2f armor_center((center1.x + center2.x) / 2,
                (center1.y + center2.y) / 2);
            armor_centers.push_back(armor_center);
        }
    }
    return armor_centers;
}

std::vector<std::vector<cv::Point>> ArmorDetector::findContoursByColor(const cv::Mat& hsv_image, const std::string& color) {
    cv::Mat mask, mask1, mask2;

    if (color == "red") {
        // 绾㈣壊鍦℉SV涓垎甯冨湪涓ょ锛岄渶瑕佷袱涓寖鍥?
        cv::inRange(hsv_image, lower_red1, upper_red1, mask1);
        cv::inRange(hsv_image, lower_red2, upper_red2, mask2);
        cv::add(mask1, mask2, mask);
    } else if (color == "blue") {
        cv::inRange(hsv_image, lower_blue, upper_blue, mask);
    }

    // 鏍规嵁鍥惧儚澶у皬璋冩暣褰㈡€佸鎿嶄綔鏍稿ぇ灏?
    int kernel_size = std::max(3, std::min(hsv_image.cols, hsv_image.rows) / 250);
    cv::Mat kernel = cv::getStructuringElement(cv::MORPH_RECT, cv::Size(kernel_size, kernel_size));
    cv::morphologyEx(mask, mask, cv::MORPH_OPEN, kernel);
    cv::morphologyEx(mask, mask, cv::MORPH_CLOSE, kernel);

    // 鏌ユ壘杞粨
    std::vector<std::vector<cv::Point>> contours;
    std::vector<cv::Vec4i> hierarchy;
    cv::findContours(mask, contours, hierarchy, cv::RETR_EXTERNAL, cv::CHAIN_APPROX_SIMPLE);

    // 鏍规嵁闈㈢Н鍜屽舰鐘惰繃婊よ疆寤?
    std::vector<std::vector<cv::Point>> filtered_contours;
    for (const auto& contour : contours) {
        double area = cv::contourArea(contour);
        if (area > min_area && area < max_area) {
            // 璁＄畻杞粨鐨勮竟鐣岀煩褰?
            cv::Rect bounding_rect = cv::boundingRect(contour);
            double ratio = static_cast<double>(bounding_rect.height) / bounding_rect.width;

            // 鏍规嵁闀垮姣旇繃婊わ紝鐏潯搴旇鏄珫鐩寸殑缁嗛暱褰㈢姸
            if (ratio > min_ratio && ratio < max_ratio) {
                filtered_contours.push_back(contour);
            }
        }
    }

    return filtered_contours;
}

void ArmorDetector::matchArmorPairs(const std::vector<std::vector<cv::Point>>& contours,
                                  const std::string& color,
                                  std::vector<ArmorInfo>& armors) {
    // 瑁呯敳鏉垮尮閰嶇畻娉曪細瀵绘壘鐩歌繎楂樺害鍜岃搴︾殑鐏潯瀵?
    for (size_t i = 0; i < contours.size(); ++i) {
        cv::RotatedRect rect1 = cv::minAreaRect(contours[i]);

        for (size_t j = i + 1; j < contours.size(); ++j) {
            cv::RotatedRect rect2 = cv::minAreaRect(contours[j]);

            // 鑾峰彇鐏潯鐨勮搴︼紙OpenCV涓搴﹁寖鍥翠负[-90,0)锛?
            float angle1 = rect1.angle;
            float angle2 = rect2.angle;

            // 鑾峰彇鐏潯鐨勪腑蹇冪偣
            cv::Point2f center1 = rect1.center;
            cv::Point2f center2 = rect2.center;

            // 璁＄畻涓ょ伅鏉′箣闂寸殑璺濈
            float distance = cv::norm(center1 - center2);

            // 淇瑙掑害宸绠楋紙鑰冭檻OpenCV瑙掑害鐗规€э級
            float angle_diff = std::abs(angle1 - angle2);
            // 瀵逛簬鎺ヨ繎-90搴︾殑鎯呭喌锛岄渶瑕佺壒娈婂鐞?
            if (angle_diff > 45) {
                angle_diff = 90 - angle_diff;
            }

            // 璁＄畻鐏潯鐨勯珮搴?
            float height1 = std::max(rect1.size.width, rect1.size.height);
            float height2 = std::max(rect2.size.width, rect2.size.height);

            // 璁＄畻楂樺害宸瘮鐜?
            float height_diff_ratio = std::abs(height1 - height2) / std::max(height1, height2) * 100;

            // 璁＄畻姘村钩璺濈涓庡瀭鐩磋窛绂绘鏌?
            float horizontal_dist = std::abs(center1.x - center2.x);
            float vertical_dist = std::abs(center1.y - center2.y);

            // 鍖归厤鏉′欢澧炲己锛?
            // 1. 瑙掑害宸皬浜庨槇鍊?
            // 2. 楂樺害宸瘮鐜囧皬浜庨槇鍊?
            // 3. 璺濈鍦ㄥ悎鐞嗚寖鍥村唴
            // 4. 姘村钩璺濈搴斿ぇ浜庡瀭鐩磋窛绂伙紙瑁呯敳鏉块€氬父鏄按骞虫帓鍒楋級
            // 5. 瀹介珮姣斿悎鐞嗘€ф鏌?
            if (angle_diff < max_angle_diff &&
                height_diff_ratio < max_height_diff_ratio &&
                distance > min_distance &&
                distance < max_distance &&
                horizontal_dist > vertical_dist) {

                // 纭畾瑁呯敳鏉跨殑杈圭晫妗?
                std::vector<cv::Point2f> points;
                cv::Point2f pts1[4], pts2[4];
                rect1.points(pts1);
                rect2.points(pts2);

                for (int k = 0; k < 4; k++) {
                    points.push_back(pts1[k]);
                }
                for (int k = 0; k < 4; k++) {
                    points.push_back(pts2[k]);
                }

                cv::Rect bbox = cv::boundingRect(points);
                cv::Point2f center = {(center1.x + center2.x) / 2, (center1.y + center2.y) / 2};

                ArmorInfo armor;
                armor.center = center;
                armor.bbox = bbox;
                armor.color = color;
                armor.confidence = (1.0f - (angle_diff / max_angle_diff)) *
                                  (1.0f - (height_diff_ratio / max_height_diff_ratio));
                armor.lightBars = points;

                // 淇濆瓨鐏潯鐨勬棆杞煩褰俊鎭?
                armor.lightRects.push_back(rect1);
                armor.lightRects.push_back(rect2);

                armors.push_back(armor);
            }
        }
    }
}

std::vector<ArmorDetector::ArmorInfo> ArmorDetector::matchArmors(
    const std::vector<std::vector<cv::Point>>& red_contours,
    const std::vector<std::vector<cv::Point>>& blue_contours) {

    std::vector<ArmorInfo> armors;

    // 澶勭悊绾㈣壊瑁呯敳鏉?
    matchArmorPairs(red_contours, "red", armors);

    // 澶勭悊钃濊壊瑁呯敳鏉?
    matchArmorPairs(blue_contours, "blue", armors);

    return armors;
}

void ArmorDetector::drawBinaryImages(const cv::Mat& hsv_image) {
    cv::Mat red_mask1, red_mask2, red_mask, blue_mask, combined_mask;

    // 绾㈣壊鎺╃爜
    cv::inRange(hsv_image, lower_red1, upper_red1, red_mask1);
    cv::inRange(hsv_image, lower_red2, upper_red2, red_mask2);
    cv::add(red_mask1, red_mask2, red_mask);

    // 钃濊壊鎺╃爜
    cv::inRange(hsv_image, lower_blue, upper_blue, blue_mask);

    // 鍚堝苟鎺╃爜鐢ㄤ簬鏄剧ず
    std::vector<cv::Mat> channels;
    channels.push_back(blue_mask);
    channels.push_back(red_mask);
    channels.push_back(cv::Mat::zeros(red_mask.size(), CV_8UC1));
    cv::merge(channels, combined_mask);

    // 鏄剧ず浜屽€煎寲鍥惧儚
    cv::imshow("浜屽€煎寲鍥惧儚", combined_mask);
}

void ArmorDetector::drawUI(cv::Mat& image, const std::vector<ArmorInfo>& armors) {
    // 缁樺埗鎵€鏈夎瘑鍒埌鐨勮鐢叉澘
    for (size_t i = 0; i < armors.size(); ++i) {
        const auto& armor = armors[i];

        // 缁樺埗瑁呯敳鏉胯竟鐣屾 (绾垮鏍规嵁鍒嗚鲸鐜囪皟鏁?
        int bbox_thickness = std::max(2, std::min(image.cols, image.rows) / 300);
        cv::rectangle(image, armor.bbox, cv::Scalar(0, 255, 0), bbox_thickness);

        // 缁樺埗涓績鐐?
        int center_radius = std::max(3, std::min(image.cols, image.rows) / 200);
        cv::circle(image, armor.center, center_radius, cv::Scalar(0, 0, 255), -1);

        // 缁樺埗鐏潯
        int light_thickness = std::max(1, std::min(image.cols, image.rows) / 400);
        for (const auto& rect : armor.lightRects) {
            cv::Point2f pts[4];
            rect.points(pts);
            for (int j = 0; j < 4; j++) {
                cv::line(image, pts[j], pts[(j+1)%4], cv::Scalar(255, 0, 0), light_thickness);
            }
        }

        // 缁樺埗璇︾粏淇℃伅闈㈡澘
        std::string color_info = "Color: " + armor.color;
        std::string coord_info = "Coord: (" + std::to_string(static_cast<int>(armor.center.x)) +
                                ", " + std::to_string(static_cast<int>(armor.center.y)) + ")";
        std::string conf_info = "Conf: " + std::to_string(armor.confidence).substr(0, 4);
        std::string id_info = "ID: " + std::to_string(i+1);

        // 鏍规嵁鍒嗚鲸鐜囪皟鏁村瓧浣撳ぇ灏忓拰闈㈡澘灏哄
        double font_scale = std::max(0.4, std::min(image.cols, image.rows) / 1000.0);
        int panel_width = static_cast<int>(150 * font_scale * 2.5);
        int panel_height = static_cast<int>(80 * font_scale * 2);
        int text_thickness = std::max(1, static_cast<int>(font_scale * 2));

        cv::Point panel_top_left(armor.bbox.x, armor.bbox.y - panel_height - 10);
        cv::Point panel_bottom_right(armor.bbox.x + panel_width, armor.bbox.y - 10);

        // 纭繚淇℃伅闈㈡澘涓嶄細瓒呭嚭鍥惧儚杈圭晫
        if (panel_top_left.x < 0) {
            panel_top_left.x = 0;
            panel_bottom_right.x = panel_width;
        }
        if (panel_bottom_right.x > image.cols) {
            panel_bottom_right.x = image.cols;
            panel_top_left.x = image.cols - panel_width;
        }
        if (panel_top_left.y < 0) {
            panel_top_left.y = armor.bbox.y + armor.bbox.height + 10;

void ArmorDetector::run() {
    if (!is_camera_opened) {
        std::cerr << "鎽勫儚澶存湭姝ｇ‘鍒濆鍖栵紝鏃犳硶杩愯绋嬪簭" << std::endl;
        return;
    }

    std::cout << "瑁呯敳鏉胯瘑鍒▼搴忓惎鍔?.." << std::endl;
    std::cout << "鎸夐敭璇存槑:" << std::endl;
    std::cout << "  q: 閫€鍑虹▼搴? << std::endl;
    std::cout << "  s: 鍒囨崲UI鏄剧ず" << std::endl;
    std::cout << "  b: 鍒囨崲浜屽€煎寲鍥惧儚鏄剧ず" << std::endl;
    std::cout << "  r: 寮€濮嬫柊鍥炲悎" << std::endl;
    std::cout << "  a: 鍏佽璇嗗埆" << std::endl;
    std::cout << "  f: 瀹屾垚璇嗗埆" << std::endl;
    std::cout << "  1: 鏄庝寒鐜蹇棬閫熷害" << std::endl;
    std::cout << "  2: 姝ｅ父鐜蹇棬閫熷害" << std::endl;
    std::cout << "  3: 榛戞殫鐜蹇棬閫熷害" << std::endl;
    std::cout << "  p: 淇濆瓨褰撳墠鍙傛暟涓洪璁? << std::endl;
    std::cout << "========================" << std::endl;

    createTrackbars();

    while (true) {
        // 鏇存柊棰滆壊鑼冨洿鍜岀浉鏈哄弬鏁帮紙浠庢粦鍔ㄦ潯鑾峰彇锛?
        updateColorRanges();
        updateCameraParams();

        // 璇诲彇甯?
        cap >> frame;
        if (frame.empty()) {
            std::cerr << "鏃犳硶鑾峰彇鎽勫儚澶寸敾闈?" << std::endl;
            break;
        }

        // 鍥惧儚棰勫鐞?
        cv::Mat processed = preprocess(frame);

        // 濡傛灉鍏佽璇嗗埆锛屽垯杩涜瑁呯敳鏉挎娴?
        std::vector<ArmorInfo> armors;
        if (is_recognition_allowed || recognition_mode == 0) {
            // 鍒嗗埆鏌ユ壘绾㈣摑瑁呯敳鏉跨殑鐏潯
            auto red_contours = findContoursByColor(processed, "red");
            auto blue_contours = findContoursByColor(processed, "blue");

            // 鍖归厤瑁呯敳鏉?
            armors = matchArmors(red_contours, blue_contours);
        }

        // 缁樺埗UI
        if (show_ui) {
            drawUI(frame, armors);
        }

        // 鏄剧ず缁撴灉
        cv::imshow("瑁呯敳鏉胯瘑鍒?, frame);

        // 鏄剧ず浜屽€煎寲鍥惧儚
        if (show_binary) {
            drawBinaryImages(processed);
        }

        // 閿洏杈撳叆澶勭悊
        char key = cv::waitKey(1);
        if (key == 'q' || key == 'Q') {
            break;
        } else if (key == 's' || key == 'S') {
            show_ui = !show_ui;
        } else if (key == 'b' || key == 'B') {
            show_binary = !show_binary;
            if (!show_binary) {
                cv::destroyWindow("浜屽€煎寲鍥惧儚");
            }
        } else if (key == 'r' || key == 'R') {
            startNewRound();
        } else if (key == 'a' || key == 'A') {
            allowRecognition();
        } else if (key == 'f' || key == 'F') {
            finishRecognition();
        } else if (key == '1') {
            setBrightCondition();
        } else if (key == '2') {
            setNormalCondition();
        } else if (key == '3') {
            setDarkCondition();
        } else if (key == 'p' || key == 'P') {
            savePresetParams();
        }
    }
}
